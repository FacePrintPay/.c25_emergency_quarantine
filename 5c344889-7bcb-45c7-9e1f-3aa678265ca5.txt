    const c = ref.current; if (!c) return;
    const ctx = c.getContext("2d");
    const resize = () => { c.width = window.innerWidth; c.height = window.innerHeight; };
    resize(); window.addEventListener("resize", resize);
    const stars = Array.from({length:220}, () => ({
      x: Math.random()*3000, y: Math.random()*3000,
      r: Math.random()*1.6+0.15, phase: Math.random()*Math.PI*2, spd: Math.random()*0.006+0.001
    }));
    let raf, t=0;
    const draw = () => {
      t+=0.016; ctx.clearRect(0,0,c.width,c.height);
      stars.forEach(s => {
        const a = 0.18+0.6*Math.abs(Math.sin(s.phase+t*s.spd*20));
        ctx.beginPath(); ctx.arc(s.x%c.width, s.y%c.height, s.r, 0, Math.PI*2);
        ctx.fillStyle=`rgba(210,225,255,${a})`; ctx.fill();
      });
      raf=requestAnimationFrame(draw);
    };
    draw();
    return ()=>{ cancelAnimationFrame(raf); window.removeEventListener("resize",resize); };
  },[]);
  return <canvas ref={ref} style={{position:"fixed",inset:0,width:"100%",height:"100%",pointerEvents:"none",zIndex:0}}/>;
}

// ─── ANIMATED PLANET CANVAS ────────────────────────────────────────────────
function AnimatedPlanet({ agent, size=110, active=false }) {
  const ref = useRef(null);
  const angleRef = useRef(Math.random()*Math.PI*2);
  useEffect(() => {
    const c = ref.current; if (!c) return;
    const ctx = c.getContext("2d");
    c.width = size; c.height = size;
    const cx = size/2, cy = size/2;
    const pr = size*0.22; // planet radius
    const orbitR = size*0.38;
    let raf;
    const draw = () => {
      ctx.clearRect(0,0,size,size);
      // Orbit ring
      ctx.beginPath();
      ctx.ellipse(cx, cy, orbitR, orbitR*0.32, -0.3, 0, Math.PI*2);
      ctx.strokeStyle = `${agent.color}28`;
      ctx.lineWidth = 1;
      ctx.stroke();

      // Saturn/Uranus ring around planet
      if (agent.ring) {
        ctx.save();
        ctx.translate(cx, cy);
        ctx.rotate(-0.3);
        ctx.scale(1, 0.32);
        ctx.beginPath();
        ctx.arc(0, 0, pr*1.7, 0, Math.PI*2);
        ctx.strokeStyle = `${agent.color}55`;
        ctx.lineWidth = 3;
        ctx.stroke();
        ctx.beginPath();
        ctx.arc(0, 0, pr*2.1, 0, Math.PI*2);
        ctx.strokeStyle = `${agent.color}33`;
        ctx.lineWidth = 1.5;
        ctx.stroke();
        ctx.restore();
      }

      // Planet body
      const grd = ctx.createRadialGradient(cx-pr*0.3, cy-pr*0.3, pr*0.05, cx, cy, pr);
      grd.addColorStop(0, agent.color+"ff");
      grd.addColorStop(0.4, agent.color+"cc");
      grd.addColorStop(1, agent.color+"22");
      ctx.beginPath(); ctx.arc(cx, cy, pr, 0, Math.PI*2);
      ctx.fillStyle = grd; ctx.fill();

      // Atmospheric glow
      const glow = ctx.createRadialGradient(cx, cy, pr*0.8, cx, cy, pr*1.5);
      glow.addColorStop(0, `${agent.color}00`);
      glow.addColorStop(1, `${agent.color}${active?"33":"18"}`);
      ctx.beginPath(); ctx.arc(cx, cy, pr*1.5, 0, Math.PI*2);
      ctx.fillStyle = glow; ctx.fill();

      // Surface details — latitude bands
      for (let i=0; i<3; i++) {
        const y = cy - pr*0.4 + i*pr*0.4;
        ctx.save();
        ctx.beginPath(); ctx.arc(cx, cy, pr, 0, Math.PI*2);
        ctx.clip();
        ctx.beginPath();
        ctx.ellipse(cx, y, pr, pr*0.08, 0, 0, Math.PI*2);
        ctx.fillStyle = `rgba(0,0,0,0.12)`;
        ctx.fill();
        ctx.restore();
      }

      // Moons orbiting
      if (agent.moons) {
        for (let m=0; m<agent.moons; m++) {
          const a = angleRef.current + (m/agent.moons)*Math.PI*2;
          const mx = cx + Math.cos(a)*orbitR;
          const my = cy + Math.sin(a)*orbitR*0.32;
          ctx.beginPath(); ctx.arc(mx, my, size*0.03, 0, Math.PI*2);
          ctx.fillStyle = `${agent.color}99`;
          ctx.fill();
        }
      }

      // Pulse ring when active
      if (active) {
        const pulse = (Date.now()%1000)/1000;
        ctx.beginPath(); ctx.arc(cx, cy, pr*(1+pulse*0.6), 0, Math.PI*2);
        ctx.strokeStyle = `${agent.color}${Math.floor((1-pulse)*60).toString(16).padStart(2,"0")}`;
        ctx.lineWidth = 1.5;
        ctx.stroke();
      }

      angleRef.current += (2*Math.PI)/(agent.speed*60);
      raf = requestAnimationFrame(draw);
    };
    draw();
    return ()=>cancelAnimationFrame(raf);
  }, [agent, active, size]);

  return <canvas ref={ref} width={size} height={size} style={{display:"block"}}/>;
}
